# 前端 AI 开发指南 - 资源模块对接

## 1. 模块概述

- **服务前缀**：`/api/resource`
- **端口**：8892
- **职责**：教育资源（图片、文档、视频）的上传、存储、管理和访问控制
- **存储方案**：腾讯云 COS（对象存储）+ 腾讯云 VOD（视频点播）

## 2. 上传流程

### 2.1 预签名上传流程

本项目采用**服务端预签名**方式，前端直传腾讯云，减轻服务器压力：

```
┌─────────┐  1. POST /resources/presign/image   ┌─────────┐
│  前端    │ ──────────────────────────────────> │  后端    │
│         │ <────────────────────────────────── │         │
│         │  { presignedUrl, resourceId }        │         │
│         │                                      │         │
│         │  2. PUT 文件到 presignedUrl          │         │
│         │ ──────────────────────────────────> │ 腾讯云   │
│         │ <────────────────────────────────── │  COS    │
│         │  200 OK                              │         │
│         │                                      │         │
│         │  3. POST /resources/confirm          │         │
│         │ ──────────────────────────────────> │  后端    │
│         │ <────────────────────────────────── │         │
│         │  { resourceId, url, ... }            │         │
└─────────┘                                      └─────────┘
```

### 2.2 视频上传差异

视频使用腾讯云 VOD（视频点播），流程类似但返回字段不同：

```
1. POST /resources/presign/video ──> 后端
2. <── { presignedUrl, resourceId, vodFileId }
3. 前端上传视频到 VOD（使用 VOD SDK 或预签名 URL）
4. POST /resources/confirm/video ──> 后端（确认上传完成）
```

## 3. API 列表

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 预签名图片 | POST | `/resources/presign/image` | 生成图片上传 URL | 已登录 |
| 预签名文档 | POST | `/resources/presign/document` | 生成文档上传 URL | 已登录 |
| 预签名视频 | POST | `/resources/presign/video` | 生成视频上传 URL | 已登录 |
| 确认上传 | POST | `/resources/confirm` | 确认 COS 上传完成 | 已登录 |
| 确认视频上传 | POST | `/resources/confirm/video` | 确认 VOD 上传完成 | 已登录 |
| 获取资源详情 | GET | `/resources/{resourceId}` | 资源详情 | 已登录 |
| 查询我的资源 | GET | `/resources` | 分页查询 | 已登录 |
| 删除资源 | DELETE | `/resources/{resourceId}` | 仅所有者可删 | 已登录 |

## 4. 前端封装

### 4.1 通用上传组件

```vue
<!-- components/UploadButton.vue -->
<template>
  <el-upload
    action="#"
    :http-request="customUpload"
    :before-upload="beforeUpload"
    :show-file-list="false"
    :accept="accept"
  >
    <el-button :loading="uploading" :type="buttonType">
      <slot>上传</slot>
    </el-button>
  </el-upload>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { ElMessage } from 'element-plus'
import { resourceApi } from '@/api/resource'

const props = defineProps<{
  resourceType: 'image' | 'document' | 'video'
  accept?: string
  buttonType?: string
  maxSize?: number  // MB
}>()

const emit = defineEmits<{
  success: [resource: { resourceId: number, url: string }]
}>()

const uploading = ref(false)

const beforeUpload = (file: File) => {
  if (props.maxSize && file.size > props.maxSize * 1024 * 1024) {
    ElMessage.error(`文件大小不能超过 ${props.maxSize}MB`)
    return false
  }
  return true
}

const customUpload = async ({ file }: { file: File }) => {
  uploading.value = true
  try {
    // 1. 获取预签名 URL
    const presignRes = await resourceApi.presign({
      resourceType: props.resourceType,
      fileName: file.name,
      fileSize: file.size,
      mimeType: file.type
    })

    // 2. 直传到腾讯云
    await fetch(presignRes.presignedUrl, {
      method: 'PUT',
      body: file,
      headers: {
        'Content-Type': file.type
      }
    })

    // 3. 确认上传
    const url = presignRes.presignedUrl.split('?')[0]
    const confirmRes = await resourceApi.confirmUpload({
      resourceId: presignRes.resourceId,
      url,
      fileName: file.name,
      fileSize: file.size
    })

    ElMessage.success('上传成功')
    emit('success', {
      resourceId: confirmRes.resourceId,
      url: confirmRes.url
    })
  } catch (error) {
    ElMessage.error('上传失败')
    throw error
  } finally {
    uploading.value = false
  }
}
</script>
```

### 4.2 资源管理页面

```vue
<template>
  <div class="resource-page">
    <div class="page-header">
      <h2>我的资源</h2>
      <el-radio-group v-model="queryParams.resourceType">
        <el-radio-button :label="undefined">全部</el-radio-button>
        <el-radio-button :label="ResourceType.IMAGE">图片</el-radio-button>
        <el-radio-button :label="ResourceType.DOCUMENT">文档</el-radio-button>
        <el-radio-button :label="ResourceType.VIDEO">视频</el-radio-button>
      </el-radio-group>
    </div>

    <div class="resource-grid">
      <el-card
        v-for="resource in resourceList"
        :key="resource.resourceId"
        class="resource-card"
      >
        <!-- 图片预览 -->
        <img v-if="resource.resourceType === ResourceType.IMAGE" :src="resource.url" />

        <!-- 视频预览 -->
        <video v-else-if="resource.resourceType === ResourceType.VIDEO" :src="resource.url" controls />

        <!-- 文档图标 -->
        <div v-else class="document-icon">
          <el-icon><Document /></el-icon>
        </div>

        <div class="resource-info">
          <p class="name">{{ resource.resourceName }}</p>
          <p class="meta">{{ formatSize(resource.size) }} · {{ formatDate(resource.createdAt) }}</p>
        </div>

        <div class="resource-actions">
          <el-button size="small" @click="copyUrl(resource.url)">复制链接</el-button>
          <el-button size="small" type="danger" @click="deleteResource(resource.resourceId)">删除</el-button>
        </div>
      </el-card>
    </div>

    <el-pagination
      v-model:current-page="queryParams.current"
      v-model:page-size="queryParams.pageSize"
      :total="total"
      @change="fetchData"
    />
  </div>
</template>

<script setup lang="ts">
import { reactive, ref, onMounted, watch } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { resourceApi } from '@/api/resource'
import { ResourceType } from '@/types/enums'
import type { ResourceVO, ResourceQueryRequest } from '@/types/resource'

const resourceList = ref<ResourceVO[]>([])
const total = ref(0)

const queryParams = reactive<ResourceQueryRequest>({
  current: 1,
  pageSize: 12,
  resourceType: undefined,
  keyword: ''
})

const fetchData = async () => {
  const res = await resourceApi.listMyResources(queryParams)
  resourceList.value = res.records
  total.value = res.total
}

const deleteResource = async (resourceId: number) => {
  await ElMessageBox.confirm('确定删除该资源吗？', '提示')
  await resourceApi.deleteResource(resourceId)
  ElMessage.success('删除成功')
  fetchData()
}

const formatSize = (bytes?: number) => {
  if (!bytes) return '-'
  const units = ['B', 'KB', 'MB', 'GB']
  let i = 0
  while (bytes >= 1024 && i < units.length - 1) {
    bytes /= 1024
    i++
  }
  return `${bytes.toFixed(2)} ${units[i]}`
}

watch(() => queryParams.resourceType, fetchData)
onMounted(fetchData)
</script>

<style scoped>
.resource-grid {
  display: grid;
  grid-template-columns: repeat(auto-fill, minmax(200px, 1fr));
  gap: 16px;
}
.resource-card img, .resource-card video {
  width: 100%;
  height: 150px;
  object-fit: cover;
}
</style>
```

## 5. API 封装

```typescript
// api/resource.ts
import request from '@/utils/request'
import type {
  ResourceVO, ResourceDetailVO, PresignedUrlVO, VodPresignedUrlVO,
  PresignRequest, UploadConfirmRequest, VideoConfirmRequest, ResourceQueryRequest
} from '@/types/resource'
import type { PageResult } from '@/types/common'
import { ResourceType } from '@/types/enums'

export const resourceApi = {
  /** 获取预签名 URL（通用） */
  presign: (data: PresignRequest & { resourceType: string }) => {
    const endpointMap: Record<string, string> = {
      image: '/resource/resources/presign/image',
      document: '/resource/resources/presign/document',
      video: '/resource/resources/presign/video'
    }
    return request.post<PresignedUrlVO | VodPresignedUrlVO>(endpointMap[data.resourceType], data)
  },

  /** 预签名图片 */
  presignImage: (data: PresignRequest) =>
    request.post<PresignedUrlVO>('/resource/resources/presign/image', data),

  /** 预签名文档 */
  presignDocument: (data: PresignRequest) =>
    request.post<PresignedUrlVO>('/resource/resources/presign/document', data),

  /** 预签名视频 */
  presignVideo: (data: PresignRequest) =>
    request.post<VodPresignedUrlVO>('/resource/resources/presign/video', data),

  /** 直传到 COS（不走 Axios 拦截器） */
  uploadToCos: (url: string, file: File) =>
    fetch(url, {
      method: 'PUT',
      body: file,
      headers: { 'Content-Type': file.type }
    }),

  /** 确认 COS 上传完成 */
  confirmUpload: (data: UploadConfirmRequest) =>
    request.post<ResourceVO>('/resource/resources/confirm', data),

  /** 确认视频上传完成 */
  confirmVideoUpload: (data: VideoConfirmRequest) =>
    request.post<ResourceDetailVO>('/resource/resources/confirm/video', data),

  /** 获取资源详情 */
  getResource: (resourceId: number) =>
    request.get<ResourceDetailVO>(`/resource/resources/${resourceId}`),

  /** 查询我的资源 */
  listMyResources: (params: ResourceQueryRequest) =>
    request.get<PageResult<ResourceVO>>('/resource/resources', { params }),

  /** 删除资源 */
  deleteResource: (resourceId: number) =>
    request.delete(`/resource/resources/${resourceId}`)
}
```
