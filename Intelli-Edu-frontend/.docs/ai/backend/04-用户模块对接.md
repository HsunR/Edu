# 前端 AI 开发指南 - 用户模块对接

## 1. 模块概述

- **服务前缀**：`/api/user`
- **端口**：8891
- **职责**：用户注册、登录、信息管理、角色分配

## 2. API 列表

### 2.1 认证接口 (AuthController)

| 接口 | 方法 | 路径 | 说明 | 需要登录 |
|------|------|------|------|----------|
| 登录 | POST | `/auth/login` | 支持用户名密码/手机验证码/邮箱验证码 | 否 |
| 注册 | POST | `/auth/register` | 用户注册 | 否 |
| 注销 | POST | `/auth/logout` | Token 加入黑名单 | 是 |
| 发送注册验证码 | POST | `/auth/register/send-code` | 发送手机/邮箱验证码 | 否 |
| 发送登录验证码 | POST | `/auth/login/send-code` | 发送手机/邮箱验证码 | 否 |
| 刷新 Token | POST | `/auth/refresh-token` | 用 Refresh Token 换取新 Access Token | 否 |

### 2.2 用户接口 (UserController)

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 获取用户信息 | GET | `/users/{userId}` | 根据 ID 获取用户信息 | 任意 |
| 查询用户列表 | GET | `/users` | 多条件分页查询 | 管理员 |
| 获取当前用户 | GET | `/users/me` | 获取登录用户详情 | 已登录 |
| 更新当前用户 | PUT | `/users/me` | 更新基本信息 | 已登录 |
| 更新头像 | PUT | `/users/me/avatar` | 更新头像 URL | 已登录 |
| 更新档案 | PUT | `/users/me/profile` | 更新教师/学生档案 | 已登录 |
| 修改密码 | PUT | `/users/me/password` | 修改密码 | 已登录 |
| 删除用户 | DELETE | `/users/{userId}` | 删除用户 | 管理员 |
| 分配教师角色 | PUT | `/users/{userId}/assign-teacher` | 提升用户为教师 | 管理员 |

## 3. 前端页面设计

### 3.1 登录页 (/login)

```vue
<template>
  <div class="login-page">
    <el-card class="login-card">
      <h2>用户登录</h2>
      <el-tabs v-model="loginType">
        <el-tab-pane label="密码登录" :name="LoginType.USERNAME_PASSWORD">
          <el-form :model="form" :rules="rules" ref="formRef">
            <el-form-item prop="username">
              <el-input v-model="form.username" placeholder="用户名/邮箱/手机号" />
            </el-form-item>
            <el-form-item prop="password">
              <el-input v-model="form.password" type="password" placeholder="密码" />
            </el-form-item>
          </el-form>
        </el-tab-pane>
        <el-tab-pane label="手机验证码" :name="LoginType.MOBILE_CODE">
          <el-form :model="form" :rules="rules" ref="formRef">
            <el-form-item prop="mobile">
              <el-input v-model="form.mobile" placeholder="手机号">
                <template #append>
                  <el-button @click="sendCode" :disabled="countdown > 0">
                    {{ countdown > 0 ? `${countdown}s` : '获取验证码' }}
                  </el-button>
                </template>
              </el-input>
            </el-form-item>
            <el-form-item prop="code">
              <el-input v-model="form.code" placeholder="验证码" />
            </el-form-item>
          </el-form>
        </el-tab-pane>
      </el-tabs>
      <el-button type="primary" @click="handleLogin" :loading="loading">登录</el-button>
      <div class="extra-links">
        <router-link to="/register">注册账号</router-link>
      </div>
    </el-card>
  </div>
</template>

<script setup lang="ts">
import { reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage } from 'element-plus'
import { useUserStore } from '@/stores/user'
import { userApi } from '@/api/user'
import { LoginType } from '@/types/enums'

const router = useRouter()
const userStore = useUserStore()
const loading = ref(false)
const countdown = ref(0)
const loginType = ref(LoginType.USERNAME_PASSWORD)

const form = reactive({
  username: '',
  password: '',
  mobile: '',
  code: '',
  loginType: LoginType.USERNAME_PASSWORD
})

const handleLogin = async () => {
  loading.value = true
  try {
    const res = await userApi.login({
      loginType: loginType.value,
      username: form.username,
      password: form.password,
      mobile: form.mobile,
      code: form.code
    })
    // 存储 Token
    localStorage.setItem('accessToken', res.accessToken)
    localStorage.setItem('refreshToken', res.refreshToken)
    // 存储用户信息
    userStore.setUserInfo({
      userId: res.userId,
      name: res.name,
      userType: res.userType
    })
    ElMessage.success('登录成功')
    router.push('/')
  } finally {
    loading.value = false
  }
}

const sendCode = async () => {
  await userApi.sendLoginCode({
    loginType: LoginType.MOBILE_CODE,
    mobile: form.mobile
  })
  ElMessage.success('验证码已发送')
  countdown.value = 60
  const timer = setInterval(() => {
    countdown.value--
    if (countdown.value <= 0) clearInterval(timer)
  }, 1000)
}
</script>
```

## 4. API 封装

```typescript
// api/user.ts
import request from '@/utils/request'
import type {
  LoginRequest, LoginResult, RegisterRequest,
  UserVO, UserDetailVO, UserQueryRequest,
  UserUpdateRequest, PasswordUpdateRequest
} from '@/types/user'
import type { PageResult } from '@/types/common'

export const userApi = {
  /** 登录 */
  login: (data: LoginRequest) =>
    request.post<LoginResult>('/user/auth/login', data),

  /** 注册 */
  register: (data: RegisterRequest) =>
    request.post('/user/auth/register', data),

  /** 注销 */
  logout: () =>
    request.post('/user/auth/logout'),

  /** 发送注册验证码 */
  sendRegisterCode: (data: { registerType: number, mobile?: string, email?: string }) =>
    request.post('/user/auth/register/send-code', data),

  /** 发送登录验证码 */
  sendLoginCode: (data: { loginType: number, mobile?: string, email?: string }) =>
    request.post('/user/auth/login/send-code', data),

  /** 刷新 Token */
  refreshToken: (refreshToken: string) =>
    request.post<LoginResult>('/user/auth/refresh-token', null, {
      params: { refreshToken }
    }),

  /** 获取当前登录用户 */
  getCurrentUser: () =>
    request.get<UserDetailVO>('/user/users/me'),

  /** 获取用户信息 */
  getUserInfo: (userId: number) =>
    request.get<UserVO>(`/user/users/${userId}`),

  /** 查询用户列表 */
  listUsers: (params: UserQueryRequest) =>
    request.get<PageResult<UserVO>>('/user/users', { params }),

  /** 更新当前用户 */
  updateUser: (data: UserUpdateRequest) =>
    request.put('/user/users/me', data),

  /** 更新头像 */
  updateAvatar: (avatarUrl: string) =>
    request.put('/user/users/me/avatar', null, {
      params: { avatarUrl }
    }),

  /** 更新档案 */
  updateProfile: (data: any) =>
    request.put('/user/users/me/profile', data),

  /** 修改密码 */
  updatePassword: (data: PasswordUpdateRequest) =>
    request.put('/user/users/me/password', data)
}
```

## 5. Pinia Store 设计

```typescript
// stores/user.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { UserType } from '@/types/enums'

export const useUserStore = defineStore('user', () => {
  // State
  const token = ref(localStorage.getItem('accessToken') || '')
  const userId = ref<number | null>(null)
  const userName = ref('')
  const userType = ref<UserType>(UserType.STUDENT)
  const avatar = ref('')

  // Getters
  const isLoggedIn = computed(() => !!token.value)
  const isTeacher = computed(() => userType.value === UserType.TEACHER)
  const isAdmin = computed(() => userType.value === UserType.ADMIN)
  const isStudent = computed(() => userType.value === UserType.STUDENT)

  // Actions
  const setUserInfo = (info: { userId: number, name: string, userType: UserType }) => {
    userId.value = info.userId
    userName.value = info.name
    userType.value = info.userType
  }

  const setToken = (newToken: string) => {
    token.value = newToken
    localStorage.setItem('accessToken', newToken)
  }

  const logout = () => {
    token.value = ''
    userId.value = null
    userName.value = ''
    userType.value = UserType.STUDENT
    localStorage.removeItem('accessToken')
    localStorage.removeItem('refreshToken')
  }

  return {
    token, userId, userName, userType, avatar,
    isLoggedIn, isTeacher, isAdmin, isStudent,
    setUserInfo, setToken, logout
  }
})
```
