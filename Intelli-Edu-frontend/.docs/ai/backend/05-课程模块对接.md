# 前端 AI 开发指南 - 课程模块对接

## 1. 模块概述

- **服务前缀**：`/api/course`
- **端口**：8893
- **职责**：课程管理、章节管理、班级管理、分类管理

## 2. API 列表

### 2.1 课程接口 (CourseController)

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 创建课程 | POST | `/courses` | 创建新课程 | 教师 |
| 更新课程 | PUT | `/courses/{courseId}` | 更新课程信息 | 教师（创建者） |
| 发布课程 | PUT | `/courses/{courseId}/publish` | 发布课程 | 教师（创建者） |
| 归档课程 | PUT | `/courses/{courseId}/archive` | 归档课程 | 教师（创建者） |
| 删除课程 | DELETE | `/courses/{courseId}` | 仅草稿可删 | 教师（创建者） |
| 我教的课程 | GET | `/courses/teaching` | 教师课程列表 | 教师 |
| 课程班级列表 | GET | `/courses/{courseId}/classes` | 查看课程下班级 | 教师 |
| 浏览公开课程 | GET | `/courses` | 公开课程列表 | 任意 |
| 课程详情 | GET | `/courses/{courseId}` | 含章节目录树 | 任意 |

### 2.2 章节接口 (ChapterController)

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 添加章节 | POST | `/courses/{courseId}/chapters` | 添加章 | 教师 |
| 更新章节 | PUT | `/chapters/{chapterId}` | 更新章标题 | 教师 |
| 删除章节 | DELETE | `/chapters/{chapterId}` | 级联删除小节 | 教师 |
| 调整章节顺序 | PUT | `/courses/{courseId}/chapters/order` | 批量排序 | 教师 |

### 2.3 小节接口 (SectionController)

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 添加小节 | POST | `/chapters/{chapterId}/sections` | 添加节 | 教师 |
| 更新小节 | PUT | `/sections/{sectionId}` | 更新节 | 教师 |
| 删除小节 | DELETE | `/sections/{sectionId}` | 级联删除资源关联 | 教师 |
| 调整小节顺序 | PUT | `/chapters/{chapterId}/sections/order` | 批量排序 | 教师 |
| 添加资源 | POST | `/sections/{sectionId}/resources` | 添加资源到节 | 教师 |
| 移除资源 | DELETE | `/sections/{sectionId}/resources/{id}` | 移除节内资源 | 教师 |
| 调整资源顺序 | PUT | `/sections/{sectionId}/resources/order` | 调整资源顺序 | 教师 |
| 获取节详情 | GET | `/sections/{sectionId}/detail` | 含资源访问地址 | 学生（需权限） |

### 2.4 分类接口 (CategoryController)

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 获取分类树 | GET | `/categories/` | 全量返回，前端缓存 | 任意 |

### 2.5 班级接口 (ClassController)

| 接口 | 方法 | 路径 | 说明 | 权限 |
|------|------|------|------|------|
| 创建班级 | POST | `/classes` | 创建班级 | 教师 |
| 更新班级 | PUT | `/classes/{classId}` | 更新班级信息 | 教师 |
| 班级成员列表 | GET | `/classes/{classId}/members` | 查看成员 | 教师 |
| 移除成员 | DELETE | `/classes/{classId}/members/{memberId}` | 移除学生 | 教师 |
| 加入班级 | POST | `/classes/join` | 通过邀请码加入 | 学生 |
| 退出班级 | POST | `/classes/{classId}/quit` | 退出班级 | 学生 |
| 我的班级 | GET | `/classes/my` | 我加入的班级 | 学生 |

## 3. 前端页面设计

### 3.1 公开课程列表 (/courses)

```vue
<template>
  <div class="course-list-page">
    <!-- 搜索和筛选 -->
    <div class="filter-bar">
      <el-input
        v-model="queryParams.keyword"
        placeholder="搜索课程"
        @keyup.enter="fetchData"
      />
      <el-select v-model="queryParams.categoryId" placeholder="全部分类" clearable>
        <el-option
          v-for="cat in categoryTree"
          :key="cat.categoryId"
          :label="cat.categoryName"
          :value="cat.categoryId"
        />
      </el-select>
      <el-button type="primary" @click="fetchData">搜索</el-button>
    </div>

    <!-- 课程卡片列表 -->
    <div class="course-grid">
      <el-card
        v-for="course in courseList"
        :key="course.courseId"
        class="course-card"
        @click="goToDetail(course.courseId)"
      >
        <img :src="course.coverUrl || defaultCover" class="course-cover" />
        <div class="course-info">
          <h3>{{ course.courseName }}</h3>
          <p class="teacher">{{ course.teacherName }}</p>
          <el-tag :type="CourseStatusTagType[course.status]">
            {{ CourseStatusMap[course.status] }}
          </el-tag>
        </div>
      </el-card>
    </div>

    <!-- 分页 -->
    <el-pagination
      v-model:current-page="queryParams.current"
      v-model:page-size="queryParams.pageSize"
      :total="total"
      @change="fetchData"
    />
  </div>
</template>

<script setup lang="ts">
import { reactive, ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { courseApi } from '@/api/course'
import type { CourseVO, CourseQueryRequest } from '@/types/course'
import type { Category } from '@/types/course'
import { CourseStatusMap, CourseStatusTagType } from '@/types/enums'

const router = useRouter()
const courseList = ref<CourseVO[]>([])
const categoryTree = ref<Category[]>([])
const total = ref(0)

const queryParams = reactive<CourseQueryRequest>({
  current: 1,
  pageSize: 12,
  keyword: '',
  categoryId: undefined,
  status: 1  // 只查已发布的
})

const fetchData = async () => {
  const res = await courseApi.listPublicCourses(queryParams)
  courseList.value = res.records
  total.value = res.total
}

const goToDetail = (courseId: number) => {
  router.push(`/courses/${courseId}`)
}

onMounted(() => {
  fetchData()
  courseApi.getCategoryTree().then(res => {
    categoryTree.value = res
  })
})
</script>
```

<template>
  <div class="teacher-course-page">

          <el-input v-model="createForm.courseName" />

        <el-form-item label="课程简介" prop="description">
          <el-input v-model="createForm.description" type="textarea" />
        </el-form-item>
        <el-form-item label="课程分类" prop="categoryId">
          <el-cascader
            v-model="createForm.categoryId"

            :props="{ value: 'categoryId', label: 'categoryName' }"
          />
        </el-form-item>
        <el-form-item label="封面">
          <el-upload
            action="#"
            :http-request="uploadCover"
            :show-file-list="false"
          >
            <img v-if="createForm.coverUrl" :src="createForm.coverUrl" class="cover-preview" />
            <el-button v-else>上传封面</el-button>
          </el-upload>
        </el-form-item>
        <el-form-item label="是否公开">
          <el-switch v-model="createForm.isPublic" :active-value="1" :inactive-value="0" />
        </el-form-item>

        <el-button @click="showCreateDialog = false">取消</el-button>

<script setup lang="ts">
import { reactive, ref, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { courseApi } from '@/api/course'
import { CourseStatus, CourseStatusMap, CourseStatusTagType } from '@/types/enums'

const courseList = ref<CourseVO[]>([])

  courseName: '',
  description: '',

  await courseApi.publishCourse(courseId)
  ElMessage.success('发布成功')
  fetchData()

  await ElMessageBox.confirm('确定删除该课程吗？', '提示')

  ElMessage.success('删除成功')
  fetchData()
}

  showCreateDialog.value = false

}

## 4. API 封装

```typescript
// api/course.ts
import request from '@/utils/request'
import type {
  CourseVO, CourseDetailVO, CourseCreateRequest, CourseUpdateRequest,
  CourseQueryRequest, ChapterVO, SectionVO, Category, Clazz, ClassVO,
  ClassMember, ChapterCreateRequest, SectionCreateRequest, JoinClassRequest
} from '@/types/course'
import type { PageResult } from '@/types/common'

export const courseApi = {
  // ===== 课程 =====
  createCourse: (data: CourseCreateRequest) =>
    request.post<CourseVO>('/course/courses', data),

  updateCourse: (courseId: number, data: CourseUpdateRequest) =>
    request.put<CourseVO>(`/course/courses/${courseId}`, data),

  publishCourse: (courseId: number) =>
    request.put(`/course/courses/${courseId}/publish`),

  archiveCourse: (courseId: number) =>
    request.put(`/course/courses/${courseId}/archive`),

  deleteCourse: (courseId: number) =>
    request.delete(`/course/courses/${courseId}`),

  listTeachingCourses: (params: CourseQueryRequest) =>
    request.get<PageResult<CourseVO>>('/course/courses/teaching', { params }),

  listPublicCourses: (params: CourseQueryRequest) =>
    request.get<PageResult<CourseVO>>('/course/courses', { params }),

  getCourseDetail: (courseId: number) =>
    request.get<CourseDetailVO>(`/course/courses/${courseId}`),

  listCourseClasses: (courseId: number) =>
    request.get<ClassVO[]>(`/course/courses/${courseId}/classes`),

  // ===== 章节 =====
  addChapter: (courseId: number, data: ChapterCreateRequest) =>
    request.post(`/course/courses/${courseId}/chapters`, data),

  updateChapter: (chapterId: number, data: { chapterName: string }) =>
    request.put(`/course/chapters/${chapterId}`, data),

  deleteChapter: (chapterId: number) =>
    request.delete(`/course/chapters/${chapterId}`),

  reorderChapters: (courseId: number, chapterIds: number[]) =>
    request.put(`/course/courses/${courseId}/chapters/order`, { chapterIds }),

  // ===== 小节 =====
  addSection: (chapterId: number, data: SectionCreateRequest) =>
    request.post(`/course/chapters/${chapterId}/sections`, data),

  updateSection: (sectionId: number, data: SectionCreateRequest) =>
    request.put(`/course/sections/${sectionId}`, data),

  deleteSection: (sectionId: number) =>
    request.delete(`/course/sections/${sectionId}`),

  reorderSections: (chapterId: number, sectionIds: number[]) =>
    request.put(`/course/chapters/${chapterId}/sections/order`, { sectionIds }),

  getSectionDetail: (sectionId: number) =>
    request.get<SectionVO>(`/course/sections/${sectionId}/detail`),

  // ===== 分类 =====
  getCategoryTree: () =>
    request.get<Category[]>('/course/categories/'),

  // ===== 班级 =====
  createClass: (data: { courseId: number, className: string, maxStudents?: number }) =>
    request.post<Clazz>('/course/classes', data),

  updateClass: (classId: number, data: Partial<Clazz>) =>
    request.put(`/course/classes/${classId}`, data),

  listMembers: (classId: number) =>
    request.get<ClassMember[]>(`/course/classes/${classId}/members`),

  removeMember: (classId: number, memberId: number) =>
    request.delete(`/course/classes/${classId}/members/${memberId}`),

  joinClass: (data: JoinClassRequest) =>
    request.post('/course/classes/join', data),

  quitClass: (classId: number) =>
    request.post(`/course/classes/${classId}/quit`),

  listMyClasses: () =>
    request.get<ClassVO[]>('/course/classes/my')
}
```
