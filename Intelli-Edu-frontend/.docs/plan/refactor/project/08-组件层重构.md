### 2.2 `DictTag/index.vue` 重写

```vue
<template>
  <template v-for="(item, index) in options" :key="index">
    <span v-if="values.includes(item.value)">
      <el-tag
        :disable-transitions="true"
        :type="item.elTagType === 'primary' ? '' : item.elTagType"
        :class="item.elTagClass"
      >
        {{ item.label }}
      </el-tag>
    </span>
  </template>
  <span v-if="unmatchedLabels.length">{{ unmatchedLabels.join(' ') }}</span>
</template>

<script setup lang="ts">
import { computed } from 'vue'

interface DictOption {
  label: string
  value: string | number
  elTagType?: '' | 'success' | 'warning' | 'danger' | 'info'
  elTagClass?: string
}

const props = defineProps<{
  options: DictOption[]
  value?: string | number | Array<string | number>
}>()

const values = computed(() => {
  if (props.value === undefined || props.value === null || props.value === '') return []
  return Array.isArray(props.value) ? props.value : [String(props.value)]
})

const unmatchedLabels = computed(() => {
  const matchedValues = props.options
    .filter(opt => values.value.includes(opt.value))
    .map(opt => opt.value)
  return values.value
    .filter(v => !matchedValues.includes(v))
    .map(v => String(v))
})
</script>
```

#### `src/directives/permission.ts`

```typescript
import type { App, Directive, DirectiveBinding } from 'vue'
import { usePermission } from '@/composables/usePermission'

function hasRole(el: HTMLElement, binding: DirectiveBinding) {
  const { hasRole } = usePermission()
  const roles = Array.isArray(binding.value) ? binding.value : [binding.value]
  if (!roles.some(role => hasRole(role))) {
    el.parentNode?.removeChild(el)
  }
}

function hasPermi(el: HTMLElement, binding: DirectiveBinding) {
  const { hasPermission } = usePermission()
  const permissions = Array.isArray(binding.value) ? binding.value : [binding.value]
  if (!permissions.some(p => hasPermission(p))) {
    el.parentNode?.removeChild(el)
  }
}

export function setupPermissionDirectives(app: App) {
  app.directive('hasRole', hasRole as Directive)
  app.directive('hasPermi', hasPermi as Directive)
}
```

#### `src/directives/copy.ts`

```typescript
import type { App, Directive, DirectiveBinding } from 'vue'
import { ElMessage } from 'element-plus'

async function copyToClipboard(text: string) {
  try {
    await navigator.clipboard.writeText(text)
    ElMessage.success('复制成功')
  } catch {
    ElMessage.error('复制失败')
  }
}

const copyDirective: Directive = {
  mounted(el: HTMLElement, binding: DirectiveBinding) {
    el._copyHandler = () => copyToClipboard(binding.value)
    el.addEventListener('click', el._copyHandler)
  },
  updated(el: HTMLElement, binding: DirectiveBinding) {
    el.removeEventListener('click', el._copyHandler)
    el._copyHandler = () => copyToClipboard(binding.value)
    el.addEventListener('click', el._copyHandler)
  },
  unmounted(el: HTMLElement) {
    el.removeEventListener('click', el._copyHandler)
    delete el._copyHandler
  },
}

export function setupCopyDirective(app: App) {
  app.directive('copy', copyDirective)
}
```

> **注意**：必须在 el 上存储 handler 引用（`el._copyHandler`），否则 `removeEventListener` 无法移除匿名箭头函数。需要扩展 HTMLElement 类型：

```typescript
declare global {
  interface HTMLElement {
    _copyHandler?: () => void
  }
}
```
