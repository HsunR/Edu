# 03 - 请求层重构

## 一、当前问题

### 1.1 `src/utils/request.ts` 问题清单

| # | 问题 | 严重程度 | 行号 |
|---|------|---------|------|
| 1 | 响应拦截器 blob/arraybuffer 分支与普通分支返回值相同，无意义 | 🟡 | L49-L55 |
| 2 | `originalRequest._retry` 在 AxiosRequestConfig 上挂载自定义属性，TS 报错 | 🔴 | L58 |
| 3 | `isRelogin` 用 `let` 导出可变对象，应该用更安全的方式 | 🟡 | L11 |
| 4 | `download` 函数参数类型全为 `any` | 🟡 | L122 |
| 5 | `message.substr()` 已废弃，应该用 `slice()` | 🟢 | L113 |
| 6 | 错误处理中 `JSON.parse(resText)` 无 try-catch | 🟡 | L139 |
| 7 | Token 刷新逻辑与 Store 耦合不清晰 | 🟡 | L70-L85 |
| 8 | 请求拦截器中 `isToken` 判断使用 `(config.headers || {}).isToken === false`，不安全 | 🟢 | L34 |

### 1.2 `src/plugins/download.js` 问题

| # | 问题 | 严重程度 |
|---|------|---------|
| 1 | 直接使用 `axios` 而非封装的 `request`，绕过了拦截器 | 🔴 |
| 2 | 手动设置 `Authorization` header，与拦截器逻辑重复 | 🟡 |
| 3 | `var` 声明 | 🟢 |
| 4 | `downloadLoadingInstance` 是模块级变量，并发调用可能互相影响 | 🟡 |

### 1.3 架构问题

当前请求层分散在三个位置：
- `src/utils/request.ts` - 主 Axios 实例
- `src/plugins/download.js` - 下载功能（独立 Axios 实例）
- `src/api/index.ts` - 旧版 API 聚合（使用旧式调用）

---

## 二、重构目标

将请求层统一到 `src/api/request.ts`，删除 `src/plugins/download.js`，所有 HTTP 请求通过统一入口发出。

```
src/api/
├── request.ts          # 🆕 统一请求层（从 utils/request.ts 迁移 + 重构）
└── ...
```

## 三、详细重构方案

### 3.1 `src/api/request.ts` - 统一请求层

```typescript
import axios, {
  type AxiosInstance,
  type AxiosRequestConfig,
  type AxiosResponse,
  type InternalAxiosRequestConfig
} from 'axios'
import { ElMessage, ElMessageBox, ElLoading } from 'element-plus'
import { getToken, getRefreshToken, setToken, setRefreshToken, removeToken } from '@/utils/auth'
import { saveAs } from 'file-saver'

const JSONbigString = JSONbig({ storeAsString: true })

let isRefreshing = false
let pendingRequests: Array<(token: string) => void> = []
let reloginShown = false

const service: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_APP_BASE_API,
  timeout: 30000,
  transformResponse: [
    (data) => {
      if (typeof data === 'string') {
        try {
          return JSONbigString.parse(data)
        } catch {
          return data
        }
      }
      return data
    }
  ]
})

// 请求拦截器
service.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = getToken()
    const noToken = config.headers?.noToken === true

    if (token && !noToken) {
      config.headers.Authorization = `Bearer ${token}`
    }

    if (config.data instanceof FormData) {
      config.headers['Content-Type'] = 'multipart/form-data'
    } else if (!config.headers['Content-Type']) {
      config.headers['Content-Type'] = 'application/json'
    }

    // 清理自定义标记，不发送到服务器
    delete config.headers.noToken

    return config
  },
  (error) => Promise.reject(error)
)

// 响应拦截器
service.interceptors.response.use(
  (response: AxiosResponse) => {
    if (response.request.responseType === 'blob' || response.request.responseType === 'arraybuffer') {
      return response.data
    }
    return response.data
  },
  async (error) => {
    const { response, config } = error

    if (!response) {
      const msg = error.message
      if (msg === 'Network Error') {
        ElMessage.error('后端接口连接异常')
      } else if (msg.includes('timeout')) {
        ElMessage.error('系统接口请求超时')
      } else {
        ElMessage.error(msg)
      }
      return Promise.reject(error)
    }

    // 401 Token 过期处理
    if (response.status === 401 && !config._retry) {
      if (isRefreshing) {
        return new Promise((resolve) => {
          pendingRequests.push((token: string) => {
            config.headers.Authorization = `Bearer ${token}`
            resolve(service(config))
          })
        })
      }

      config._retry = true
      isRefreshing = true

      try {
        const refreshToken = getRefreshToken()
        if (!refreshToken) throw new Error('No refresh token')

        const res = await axios.post(
          `${import.meta.env.VITE_APP_BASE_API}/api/user/auth/refresh-token`,
          null,
          { params: { refreshToken } }
        )

        const data = res.data.data || res.data
        setToken(data.accessToken)
        setRefreshToken(data.refreshToken)

        pendingRequests.forEach(cb => cb(data.accessToken))
        pendingRequests = []

        config.headers.Authorization = `Bearer ${data.accessToken}`
        return service(config)
      } catch {
        if (!reloginShown) {
          reloginShown = true
          ElMessageBox.confirm('登录状态已过期，请重新登录', '系统提示', {
            confirmButtonText: '重新登录',
            cancelButtonText: '取消',
            type: 'warning'
          })
            .then(() => {
              removeToken()
              location.href = '/login'
            })
            .catch(() => {})
            .finally(() => {
              reloginShown = false
            })
        }
        return Promise.reject(new Error('登录已过期，请重新登录'))
      } finally {
        isRefreshing = false
      }
    }

    // 其他错误
    const statusMessages: Record<number, string> = {
      400: '请求参数错误',
      403: '没有权限访问',
      404: '请求资源不存在',
      500: '服务器内部错误',
      502: '网关错误',
      503: '服务不可用',
    }

    const message = statusMessages[response.status]
      || `系统接口${String(response.status)}异常`
    ElMessage.error(message)

    return Promise.reject(error)
  }
)

// 扩展 AxiosRequestConfig 类型
declare module 'axios' {
  interface AxiosRequestConfig {
    _retry?: boolean
    noToken?: boolean
  }
}

// 通用请求方法
export function get<T>(url: string, params?: Record<string, unknown>, config?: AxiosRequestConfig): Promise<T> {
  return service.get(url, { params, ...config })
}

export function post<T>(url: string, data?: unknown, config?: AxiosRequestConfig): Promise<T> {
  return service.post(url, data, config)
}

export function put<T>(url: string, data?: unknown, config?: AxiosRequestConfig): Promise<T> {
  return service.put(url, data, config)
}

export function del<T>(url: string, config?: AxiosRequestConfig): Promise<T> {
  return service.delete(url, config)
}

// 文件下载
export function download(url: string, params: Record<string, unknown>, filename: string) {
  const loadingInstance = ElLoading.service({
    text: '正在下载数据，请稍候',
    background: 'rgba(0, 0, 0, 0.7)'
  })

  return service
    .post(url, params, {
      responseType: 'blob',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' }
    })
    .then(async (data: Blob) => {
      if (data.type && data.type.includes('application/json')) {
        const text = await data.text()
        try {
          const error = JSON.parse(text)
          ElMessage.error(error.message || '下载失败')
        } catch {
          ElMessage.error('下载文件出现错误')
        }
      } else {
        const blob = new Blob([data])
        saveAs(blob, filename)
      }
    })
    .catch(() => {
      ElMessage.error('下载文件出现错误，请联系管理员')
    })
    .finally(() => {
      loadingInstance.close()
    })
}

export default service
```

### 3.2 API 文件调用方式变更

所有 API 文件改为从 `@/api/request` 导入，使用快捷方法：

```typescript
// 旧版（删除）
import request from '@/utils/request'
export function login(data: LoginRequest) {
  return request.post<any, LoginResult>('/api/user/auth/login', data)
}

// 新版
import { post } from '@/api/request'
import type { LoginRequest, LoginResult } from './types'

export function login(data: LoginRequest) {
  return post<LoginResult>('/api/user/auth/login', data)
}
```

### 3.3 删除 `src/plugins/download.js`

下载功能已合并到 `api/request.ts` 的 `download` 函数中。

所有使用 `download` 的地方改为：

```typescript
// 旧版
import { download } from '@/utils/request'
// 或
this.download(url, params, filename)

// 新版
import { download } from '@/api/request'
download(url, params, filename)
```
